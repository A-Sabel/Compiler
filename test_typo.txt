// --- LEVEL 1: The Basics (Simple Variables & Types) ---
void levelOne() {
    int a = 10;

    int sum = a + b;           // PASS: Standard math is legal
    
    int a = 30;                // ERROR: Variable 'a' is already defined in this scope
    c = 50;                    // ERROR: Variable 'c' used in assignment before declaration

// --- LEVEL 2: Operators & Logic (Medium) ---
void levelTwo() {
    boolean isTrue = true;
    boolean isFalse = false;
    
    boolean logic = isTrue && isFalse; // PASS: Boolean logic
    boolean badLogic = isTrue && 10;   // ERROR: Operator '&&' requires boolean operands
    
    String concat = "Score: " + 100;   // PASS: String concatenation allows mixed types
    int badMath = "Score" * 10;        // ERROR: Operator '*' cannot be applied to 'String' and 'int'
    
    int division = 100 / 0;            // ERROR: Arithmetic Exception: / by zero
}

// --- LEVEL 3: Control Flow & Scoping (Medium-Hard) ---
void levelThree() {
    int outer = 5;
    
    for (int i = 0; i < 3; i = i + 1) {
        int inner = outer * i; // PASS: Accessing outer scope from inside loop
        outer = inner;         // PASS: Modifying outer scope
    }
    
    inner = 10;                // ERROR: Variable 'inner' used before declaration (it died in the loop)
    int i = 99;                // PASS: 'i' died in the loop, so redeclaring it here is totally legal
}
