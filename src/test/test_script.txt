class FibonacciTest {
    void main() {
        int n = 10;
        int a = 0;
        int b = 1;
        int i = 0;
        int temp;

        print("Fibonacci sequence up to " + n + " :");

        while (i < n) {
            print(a);
            temp = a + b;
            a = b;
            b = temp;
            i = i + 1;
        }
    }
}

class ScopeTest {
    // x gets Slot 0, y gets Slot 1
    int calculate(int x, int y) {
        int z = 100; // z gets Slot 2
        return (x * y) + z;
    }

    void main() {
        int a = 5;  // a gets Slot 0
        int b = 10; // b gets Slot 1
        
        // The VM must create a new frame/local array for calculate()
        int result = calculate(a, b); // result gets Slot 2
        
        print(result); // Expected Output: 150
    }
}

class ControlFlowTest {
    void main() {
        int score = 85;
        
        if (score >= 90) {
            print("Grade: A");
        } else {
            if (score >= 80) {
                if (score == 85) {
                    print("Grade: B+"); 
                } else {
                    print("Grade: B");
                }
            } else {
                print("Grade: F");
            }
        }
    }
}

class Demo {
    void main() {
        int x = 7;

        println(msg + " started...");

            println(x + " is even");
        } else {
            println(x + " is odd");
        }

        // Converted to a while loop just in case 'for' isn't supported yet

        }

        println("Done!");
    }
}

class ArraySumTest {
    void main() {
        // Allocate array of size 5
        int[] numbers = new int[5]; 
        
        // Store values (IASTORE)
        numbers[0] = 10;
        numbers[1] = 20;
        numbers[2] = 30;
        numbers[3] = 40;
        numbers[4] = 50;

        int sum = 0;
        int i = 0;

        // Load values (IALOAD)
        while (i < 5) {
            sum = sum + numbers[i];
            println("Added " + numbers[i] + ", current sum: " + sum);
            i = i + 1;
        }

        println("Total Array Sum: " + sum);
    }
}
