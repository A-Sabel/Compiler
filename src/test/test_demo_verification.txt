class DemoVerification {
    int add(int a, int b) {
        return a + b;
    }

    void main() {
        System.out.println("--- Demo Verification ---");

        int total = add(3, 4);
        int[] numbers = new int[3];
        numbers[0] = 10;
        numbers[1] = 20;
        numbers[2] = 30;

        int i = 0;
        while (i < 3) {
            total = total + numbers[i];
            i = i + 1;
        }

        int day = 2;
        switch (day) {
            case 1:
                System.out.println("Monday");
                break;
            case 2:
                System.out.println("Tuesday");
                break;
            default:
                System.out.println("Other day");
        }

        String label = (total > 0) ? "positive" : "non-positive";
        System.out.println("Label: " + label);
        System.out.println("Final total: " + total);
    }
}
